\begin{semanticmodule}
theorem{
  name: countPositive,
  parameters: [rhythm : Braid.Breath::Rhythm, position : Nat],
  statement: ((position < (Braid.Breath::cycleLength rhythm)) -> (1 <= (Braid.Breath::countAt rhythm position))),
  proof: linear_arithmetic{definitions: [Braid.Breath::countAt, Braid.Breath::cycleLength]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: countWithinPhase,
  parameters: [rhythm : Braid.Breath::Rhythm, position : Nat],
  statement: ((Braid.Breath::countAt rhythm position) <= (Braid.Breath::cycleLength rhythm)),
  proof: linear_arithmetic{definitions: [Braid.Breath::countAt, Braid.Breath::cycleLength]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: phaseBounded,
  parameters: [rhythm : Braid.Breath::Rhythm, position : Nat],
  statement: ((Braid.Breath::phaseAt rhythm position) <= 3),
  proof: linear_arithmetic{definitions: [Braid.Breath::phaseAt]},
  axioms: [Quot.sound, propext]
}
\end{semanticmodule}
