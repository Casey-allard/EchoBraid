\begin{lexlean}{Braid.Resonance}
\useglossary{lexlean.std.bool@1.2.0}
\useglossary{lexlean.std.nat@1.2.0}
\useglossary{prism.arch@1.0.0}
\useglossary{echobraid.domain@1.0.0}
\title{EchoBraid learning companion model}
\begin{semanticmodule}
definition{name: driftCeiling, parameters: [], result: Nat, body: 100}

definition{
  name: clampDrift,
  parameters: [value : Nat],
  result: Nat,
  body: (if (value <=? 100) then value else 100)
}

definition{
  name: driftStep,
  parameters: [drift : Nat, length : Nat],
  result: Nat,
  body: (if (100 <? length) then (clampDrift (drift + 10)) else (if (length <? 20) then (prim subtract (clampDrift drift) 10 : Nat) else (clampDrift drift)))
}

definition{
  name: restStep,
  parameters: [drift : Nat],
  result: Nat,
  body: (prim subtract (clampDrift drift) 30 : Nat)
}

definition{
  name: coherenceRank,
  parameters: [drift : Nat],
  result: Nat,
  body: (if (drift <? 30) then 0 else (if (drift <? 70) then 1 else 2))
}

definition{name: isDivergent, parameters: [drift : Nat], result: Bool, body: (70 <=? drift)}

definition{name: offersPause, parameters: [drift : Nat], result: Bool, body: (80 <? drift)}

definition{
  name: coherenceName,
  parameters: [drift : Nat],
  result: String,
  body: (if ((coherenceRank drift) =? 0) then "stable" else (if ((coherenceRank drift) =? 1) then "drifting" else "divergent"))
}
\end{semanticmodule}
\end{lexlean}
