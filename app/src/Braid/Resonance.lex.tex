\begin{lexlean}{Braid.Resonance}
\useglossary{lexlean.std.bool@1.2.0}
\useglossary{lexlean.std.nat@1.2.0}
\useglossary{prism.arch@1.0.0}
\useglossary{echobraid.domain@1.0.0}
\title{EchoBraid learning companion model}
\begin{semanticmodule}
definition{name: driftCeiling, parameters: [], result: Nat, body: 100}

definition{
  name: clampDrift,
  parameters: [value : Nat],
  result: Nat,
  body: (if (value <=? 100) then value else 100)
}

definition{
  name: driftStep,
  parameters: [drift : Nat, length : Nat],
  result: Nat,
  body: (if (100 <? length) then (clampDrift (drift + 10)) else (if (length <? 20) then (prim subtract (clampDrift drift) 10 : Nat) else (clampDrift drift)))
}

definition{
  name: restStep,
  parameters: [drift : Nat],
  result: Nat,
  body: (prim subtract (clampDrift drift) 30 : Nat)
}

definition{
  name: coherenceRank,
  parameters: [drift : Nat],
  result: Nat,
  body: (if (drift <? 30) then 0 else (if (drift <? 70) then 1 else 2))
}

definition{name: isDivergent, parameters: [drift : Nat], result: Bool, body: (70 <=? drift)}

definition{name: offersPause, parameters: [drift : Nat], result: Bool, body: (80 <? drift)}

definition{
  name: coherenceName,
  parameters: [drift : Nat],
  result: String,
  body: (if ((coherenceRank drift) =? 0) then "stable" else (if ((coherenceRank drift) =? 1) then "drifting" else "divergent"))
}

theorem{
  name: driftStepBounded,
  parameters: [drift : Nat, length : Nat],
  statement: ((driftStep drift length) <= 100),
  proof: linear_arithmetic{definitions: [clampDrift, driftStep]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: restStepBounded,
  parameters: [drift : Nat],
  statement: ((restStep drift) <= 100),
  proof: linear_arithmetic{definitions: [clampDrift, restStep]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: restNeverRaisesDrift,
  parameters: [drift : Nat],
  statement: ((drift <= 100) -> ((restStep drift) <= drift)),
  proof: linear_arithmetic{definitions: [clampDrift, restStep]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: pauseOnlyWhenDivergent,
  parameters: [drift : Nat],
  statement: (((offersPause drift) = true) -> ((isDivergent drift) = true)),
  proof: linear_arithmetic{definitions: [isDivergent, offersPause]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: divergentDecides,
  parameters: [drift : Nat],
  statement: (((isDivergent drift) = true) <-> ((coherenceRank drift) = 2)),
  proof: linear_arithmetic{definitions: [coherenceRank, isDivergent]},
  axioms: [Classical.choice, Quot.sound, propext]
}

theorem{
  name: coherenceMonotone,
  parameters: [lower : Nat, upper : Nat],
  statement: ((lower <= upper) -> ((coherenceRank lower) <= (coherenceRank upper))),
  proof: linear_arithmetic{definitions: [coherenceRank]},
  axioms: [Quot.sound, propext]
}
\end{semanticmodule}
\end{lexlean}
