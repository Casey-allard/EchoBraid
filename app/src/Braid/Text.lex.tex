\title{EchoBraid learning companion model}
\begin{semanticmodule}
definition{
  name: digit,
  parameters: [value : Nat],
  result: String,
  body: (if (value =? 0) then "0" else (if (value =? 1) then "1" else (if (value =? 2) then "2" else (if (value =? 3) then "3" else (if (value =? 4) then "4" else (if (value =? 5) then "5" else (if (value =? 6) then "6" else (if (value =? 7) then "7" else (if (value =? 8) then "8" else "9")))))))))
}

definition{
  name: decimalFrom,
  parameters: [fuel : Nat, value : Nat],
  result: String,
  body: (match fuel
    | Nat.zero => (digit (prim remainder value 10 0 : Nat))
    | Nat.succ rest => (if (value <? 10) then (digit value) else (prim join (cons (decimalFrom rest (prim quotient value 10 0 : Nat)) (cons (digit (prim remainder value 10 0 : Nat)) (nil String))) "" : String))),
  axioms: [propext],
  recursive_argument: fuel
}

definition{
  name: decimal,
  parameters: [value : Nat],
  result: String,
  body: (decimalFrom 6 value),
  axioms: [propext]
}

definition{
  name: fields,
  parameters: [values : (List String)],
  result: Bytes,
  body: (prim utf8_encode (prim join values "\t" : String) : Bytes),
  axioms: [propext]
}
\end{semanticmodule}
\end{lexlean}
