definition{
  name: guideIndex,
  parameters: [drift : Nat, variant : Nat],
  result: Nat,
  body: (if (Braid.Resonance::offersPause drift) then 0 else (1 + ((prim multiply (Braid.Resonance::coherenceRank drift) 3 : Nat) + (prim remainder variant 3 0 : Nat))))
}
